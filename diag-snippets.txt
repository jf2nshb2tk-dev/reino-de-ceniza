function decSegD(px,pz,ax,az,bx,bz){const dx=bx-ax,dz=bz-az,l=dx*dx+dz*dz;let t=l?((px-ax)*dx+(pz-az)*dz)/l:0;t=t<0?0:t>1?1:t;const qx=ax+dx*t-px,qz=az+dz*t-pz;return Math.sqrt(qx*qx+qz*qz);}

}

const _dM=new THREE.Matrix4(),_dQ=new THREE.Quaternion(),_dP=new THREE.Vector3(),_dS=new THREE.Vector3(),_dE=new THREE.Euler(),_dC=new THREE.Color();
function decInst(g,geo,mat,list,shadow){
  const im=new THREE.InstancedMesh(geo,mat,Math.max(1,list.length));
  for(let i=0;i<list.length;i++){
    const o=list[i];_dE.set(o.rx||0,o.ry||0,o.rz||0);_dQ.setFromEuler(_dE);_dP.set(o.x,o.y,o.z);
    const s=o.s||1;_dS.set(o.sx||s,o.sy||s,o.sz||s);_dM.compose(_dP,_dQ,_dS);im.setMatrixAt(i,_dM);
    if(o.c!==undefined){_dC.setHex(o.c);im.setColorAt(i,_dC);}
  }
  im.count=list.length;im.instanceMatrix.needsUpdate=true;if(im.instanceColor)im.instanceColor.needsUpdate=true;
  im.frustumCulled=false;im.castShadow=!!shadow&&Q>0;im.receiveShadow=true;g.add(im);return im;

function decMerge(parts){
  const pos=[],nor=[],col=[],uv=[],idx=[],m=new THREE.Matrix4(),nm=new THREE.Matrix3(),v=new THREE.Vector3(),c=new THREE.Color();let off=0;
  parts.forEach(pt=>{
    m.compose(new THREE.Vector3(pt.x||0,pt.y||0,pt.z||0),new THREE.Quaternion().setFromEuler(new THREE.Euler(pt.rx||0,pt.ry||0,pt.rz||0)),new THREE.Vector3(pt.sx||1,pt.sy||1,pt.sz||1));
    nm.getNormalMatrix(m);c.setHex(pt.c);
    const p=pt.g.attributes.position,n=pt.g.attributes.normal,u=pt.g.attributes.uv;
    for(let i=0;i<p.count;i++){
      v.fromBufferAttribute(p,i).applyMatrix4(m);pos.push(v.x,v.y,v.z);
      v.fromBufferAttribute(n,i).applyMatrix3(nm).normalize();nor.push(v.x,v.y,v.z);
      col.push(c.r,c.g,c.b);uv.push(u?u.getX(i):0,u?u.getY(i):0);
    }
    const ix=pt.g.index.array;for(let i=0;i<ix.length;i++)idx.push(ix[i]+off);off+=p.count;
  });
  const g=new THREE.BufferGeometry();
  g.setAttribute('position',new THREE.Float32BufferAttribute(pos,3));g.setAttribute('normal',new THREE.Float32BufferAttribute(nor,3));
  g.setAttribute('color',new THREE.Float32BufferAttribute(col,3));g.setAttribute('uv',new THREE.Float32BufferAttribute(uv,2));g.setIndex(idx);
  return g;
}

    const p=pt.g.attributes.position,n=pt.g.attributes.normal,u=pt.g.attributes.uv;

      v.fromBufferAttribute(n,i).applyMatrix3(nm).normalize();nor.push(v.x,v.y,v.z);
      col.push(c.r,c.g,c.b);uv.push(u?u.getX(i):0,u?u.getY(i):0);
    }
    const ix=pt.g.index.array;for(let i=0;i<ix.length;i++)idx.push(ix[i]+off);off+=p.count;
  });
  const g=new THREE.BufferGeometry();

  g.setAttribute('color',new THREE.Float32BufferAttribute(col,3));g.setAttribute('uv',new THREE.Float32BufferAttribute(uv,2));g.setIndex(idx);

function decPts(mi,n,seed,o){
  o=o||{};const R=rngMap(seed),out=[],bs=MAP_BOSSES[mi]||[],ext=o.ext||(LIMIT-4);let t=0;
  while(out.length<n&&t<n*40){

    else{x=(R()*2-1)*ext;z=(R()*2-1)*ext;
      if(Math.hypot(x,z-6)<(o.safe||17))continue;
      if(bs.some(b=>Math.hypot(x-b.x,z-b.z)<(o.arena||15)))continue;
      if(!o.onPath&&decPathD(mi,x,z)<(o.pw===undefined?4.5:o.pw))continue;

    out.push({x:x,z:z,r:R()});
  }
